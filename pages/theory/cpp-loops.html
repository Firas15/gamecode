C++ Практик

[ ЦИКЛ FOR ]

Самый распространённый цикл в C++. Три части: инициализация, условие, шаг.

C++
// for (нач; условие; шаг)
for (int i = 0; i < 5; i++) {{
    cout << i << " ";
}}
// 0 1 2 3 4

// Перебор массива
int scores[] = {{10, 20, 30}};
for (int i = 0; i < 3; i++) {{
    cout << scores[i] << endl;
}}

// Range-based for (C++11)
for (int s : scores) {{
    cout << s << endl;
}}

[ WHILE И DO-WHILE ]

C++
// while: сначала проверка
int hp = 100;
while (hp > 0) {{
    cout << "HP: " << hp << endl;
    hp -= 30;
}}

// do-while: сначала выполнение, потом проверка
int n;
do {{
    cout << "Введи число > 0: ";
    cin >> n;
}} while (n <= 0);

[ BREAK И CONTINUE ]

C++
for (int i = 0; i < 10; i++) {{
    if (i % 2 == 0) continue;  // пропускаем чётные
    if (i == 7)       break;     // стоп на 7
    cout << i << " ";
}}
// 1 3 5
