Целочисленное деление! В C++ 7 / 2 = 3 (не 3.5!), потому что оба числа целые. Для дробного результата: 7.0 / 2 или (double)7 / 2.

[ ОПЕРАТОРЫ ИНКРЕМЕНТА ]

C++
int x = 5;
x++;       // x = x + 1  → 6  (постфикс)
++x;       // x = x + 1  → 7  (префикс)
x--;       // x = x - 1  → 6

x += 10;  // x = x + 10 → 16
x *= 2;   // x = x * 2  → 32
x /= 4;   // x = x / 4  → 8
x %= 3;   // x = x % 3  → 2

[ ОПЕРАТОРЫ СРАВНЕНИЯ ]

C++
cout << (5 == 5);  // 1 (true)
cout << (5 != 3);  // 1 (true)
cout << (5 >  3);  // 1 (true)
cout << (5 <  3);  // 0 (false)
cout << (5 >= 5);  // 1 (true)

// Логические:
cout << (5>3 && 2<4);  // AND → 1
cout << (5>9 || 2<4);  // OR  → 1
cout << !true;           // NOT → 0
